import assert from 'node:assert/strict';
const {chromium}=await import(process.env.PLAYWRIGHT_MODULE||'playwright');
const browser=await chromium.launch({headless:true,executablePath:process.env.CHROME_PATH});
try{const context=await browser.newContext();const errors=[];await context.route('**/*',async r=>{const u=new URL(r.request().url());if(u.pathname==='/config.json'){const res=await r.fetch();return r.fulfill({json:{...await res.json(),cloudEnabled:false}});}if(u.hostname!=='127.0.0.1')return r.abort();return r.continue();});const p=await context.newPage();p.on('pageerror',e=>errors.push(e.message));
for(const width of [320,390,768,1440]){await p.setViewportSize({width,height:900});for(const path of ['/','/privacy.html','/terms.html','/contact.html','/admin.html']){await p.goto('http://127.0.0.1:8767'+path);if(path==='/')await p.waitForFunction(()=>!document.querySelector('#use-location').disabled);assert.ok(await p.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));assert.equal(await p.locator('a[href^="https://buy.stripe.com/"]').count(),4);}}
await p.goto('http://127.0.0.1:8767');await p.waitForFunction(()=>!document.querySelector('#use-location').disabled);assert.equal(await p.locator('.place-card').count(),0);assert.equal(await p.locator('[data-year]').innerText(),String(new Date().getFullYear()));assert.deepEqual(errors,[]);await p.setViewportSize({width:390,height:844});await p.locator('#explore').scrollIntoViewIfNeeded();await p.screenshot({path:'work/foodfinder-simple-mobile.png'});console.log('PASS: clean initial state, 20 responsive page sizes, footer support links, current year, no page errors.');}finally{await browser.close();}
